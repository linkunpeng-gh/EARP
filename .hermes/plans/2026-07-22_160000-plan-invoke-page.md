# Plan & Invoke 页面实施计划

> **For Hermes:** 按任务顺序执行，纯静态前端实现。Phase 1 用模拟数据验证交互，Phase 2 接线真实 API。

**Goal:** 重写 Plan & Invoke 页面——左侧历史面板 + 右侧调试主区，实现 Intent→Plan→Steps→Invoke→Result 完整交互流。

**Architecture:** 纯 HTML + CSS + vanilla JS。左侧面板用 CSS grid 分栏，Steps 行内展开用 CSS class toggle。JS 管理所有状态：session/history/steps/results/errors/loading。

**Tech Stack:** HTML + CSS (admin.css 复用) + vanilla JS (EARP.fetchJSON)

### Task 3: plan.html — JS 状态管理 + 模拟数据

**Objective:** 在 `<script>` 标签中实现完整交互逻辑——历史列表渲染、Plan 执行、Steps 生成、Invoke 行内展开、错误处理、Loading 状态。

**Files:**
- Modify: `apps/earp-admin/pages/plan.html`（在 `</main>` 后追加 `<script>`）

**JS 代码框架：**

```html
<script>
// ── State ──
var state = {
  currentSessionId: null,
  sessions: [],     // [{id, status, user_id, role_id, created_at, last_plan_intent, last_plan_steps}]
  steps: [],        // [{step_id, capability_id, input, result, error, loading}]
  planLoading: false,
};

// ── Demo data ──
var demoSessions = [
  {id:'sess-abc123',status:'active',user_id:'u1',role_id:'r1',created_at:'2026-07-21T10:00:00Z',
   last_plan_intent:'echo hello',last_plan_steps:2},
  {id:'sess-def456',status:'active',user_id:'u2',role_id:'r1',created_at:'2026-07-20T09:30:00Z',
   last_plan_intent:'query all users',last_plan_steps:1},
  {id:'sess-ghi789',status:'active',user_id:'u1',role_id:'r2',created_at:'2026-07-22T08:00:00Z',
   last_plan_intent:'send email to admin',last_plan_steps:3},
];

// ── Init ──
document.addEventListener('DOMContentLoaded', function() {
  loadHistory();
  // Default: auto-create mode
  updateSessionCtx(null);
});

// ── History ──
function loadHistory() {
  state.sessions = demoSessions; // TODO: EARP.fetchJSON('v1/sessions?...')
  renderHistory();
  if (state.sessions.length > 0 && !state.currentSessionId) {
    selectSession(state.sessions[0].id);
  }
}

function renderHistory() {
  var list = document.getElementById('history-list');
  if (state.sessions.length === 0) {
    list.innerHTML = '<p class="plan-empty">No sessions yet.</p>';
    return;
  }
  list.innerHTML = state.sessions.map(function(s) {
    var active = s.id === state.currentSessionId ? ' active' : '';
    var summary = s.last_plan_intent || 'No plan yet';
    var steps = s.last_plan_steps ? s.last_plan_steps + ' steps' : '';
    return '<div class="history-item' + active + '" onclick="selectSession(\'' + s.id + '\')">'
      + '<div class="hi-id">' + s.id + '</div>'
      + '<div class="hi-summary">' + summary + '</div>'
      + '<div class="hi-meta"><span>' + steps + '</span></div>'
      + '</div>';
  }).join('');
}

function selectSession(id) {
  state.currentSessionId = id;
  state.steps = [];
  renderHistory();
  renderSteps();
  var s = state.sessions.find(function(x){return x.id===id;});
  updateSessionCtx(s);
}

function newSession() {
  state.currentSessionId = null;
  state.steps = [];
  renderHistory();
  renderSteps();
  updateSessionCtx(null);
  hideError();
}

function updateSessionCtx(s) {
  var ctx = document.getElementById('session-ctx');
  if (!s) {
    ctx.style.display = 'none';
    return;
  }
  ctx.style.display = 'flex';
  document.getElementById('sc-id').textContent = s.id;
  document.getElementById('sc-status').textContent = s.status;
  document.getElementById('sc-user').textContent = s.user_id;
  document.getElementById('sc-role').textContent = s.role_id;
  document.getElementById('sc-last-plan').textContent = s.last_plan_intent
    ? s.last_plan_intent + ' (' + s.last_plan_steps + ' steps)' : '—';
}

// ── Plan ──
function doPlan() {
  var intent = document.getElementById('intent-input').value.trim();
  if (!intent) return;

  // Auto-create session if needed
  if (!state.currentSessionId) {
    var newId = 'sess-' + Date.now().toString(36);
    state.currentSessionId = newId;
    state.sessions.unshift({
      id: newId, status: 'active', user_id: 'u1', role_id: 'r1',
      created_at: new Date().toISOString(), last_plan_intent: null, last_plan_steps: null
    });
    renderHistory();
    updateSessionCtx(state.sessions[0]);
  }

  setPlanLoading(true);
  hideError();

  // Simulate Plan API call
  setTimeout(function() {
    // Simulate LLM response
    var demoSteps = [
      {step_id:1, capability_id:'cap-demo-echo', input:{msg:intent}},
      {step_id:2, capability_id:'cap-query-users', input:{limit:5}},
    ];
    state.steps = demoSteps;

    // Update session last_plan
    var s = state.sessions.find(function(x){return x.id===state.currentSessionId;});
    if (s) {
      s.last_plan_intent = intent;
      s.last_plan_steps = demoSteps.length;
    }

    setPlanLoading(false);
    renderSteps();
    renderHistory();
    updateSessionCtx(s);
  }, 800);
}

function setPlanLoading(loading) {
  state.planLoading = loading;
  var btn = document.getElementById('plan-btn');
  if (loading) {
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Planning...';
  } else {
    btn.disabled = false;
    btn.textContent = 'Plan';
  }
}

// ── Steps ──
function renderSteps() {
  var tbody = document.getElementById('steps-tbody');
  var table = document.getElementById('steps-table');
  var empty = document.getElementById('plan-empty');

  if (!state.steps || state.steps.length === 0) {
    table.style.display = 'none';
    empty.style.display = 'block';
    return;
  }
  table.style.display = '';
  empty.style.display = 'none';

  tbody.innerHTML = state.steps.map(function(s, i) {
    var inputStr = JSON.stringify(s.input);
    var loadingAttr = s.loading ? ' disabled' : '';
    var btnText = s.loading
      ? '<span class="spinner"></span>Invoking...' : 'Invoke';
    var resultHtml = '';
    if (s.result) {
      resultHtml = '<tr class="step-result-row"><td colspan="4">'
        + '<div class="step-result expanded success">' + JSON.stringify(s.result, null, 2) + '</div>'
        + '</td></tr>';
    } else if (s.error) {
      resultHtml = '<tr class="step-result-row"><td colspan="4">'
        + '<div class="step-result expanded error">' + s.error + '</div>'
        + '</td></tr>';
    }
    return '<tr>'
      + '<td>' + (i+1) + '</td>'
      + '<td><code>' + s.capability_id + '</code></td>'
      + '<td><code>' + inputStr.substring(0, 60) + (inputStr.length>60?'...':'') + '</code></td>'
      + '<td><button class="btn-sm" onclick="doInvoke(' + i + ')"' + loadingAttr + '>' + btnText + '</button></td>'
      + '</tr>' + resultHtml;
  }).join('');
}

// ── Invoke ──
function doInvoke(index) {
  var step = state.steps[index];
  step.loading = true;
  step.result = null;
  step.error = null;
  renderSteps();

  // Simulate Invoke
  setTimeout(function() {
    step.loading = false;
    // Simulate success for odd-index steps, error for even-index steps (demo)
    if (Math.random() > 0.3) {
      step.result = {echo: {message: 'Processed: ' + JSON.stringify(step.input)}};
    } else {
      step.error = 'Invoke failed: timeout after 30s';
    }
    renderSteps();
  }, 600);
}

// ── Error ──
function showError(msg) {
  var el = document.getElementById('error-banner');
  el.textContent = msg;
  el.classList.add('visible');
}
function hideError() {
  var el = document.getElementById('error-banner');
  el.classList.remove('visible');
}
</script>
```

### Task 4: 接线真实 API

**Objective:** 将 JS 中的模拟数据替换为 `EARP.fetchJSON()` 调用。

**Files:**
- Modify: `apps/earp-admin/pages/plan.html`（替换 JS 中的 demo 部分）

**变更点：**

1. `loadHistory()` → 替换 `state.sessions = demoSessions` 为：
```js
EARP.fetchJSON('v1/sessions?status=active&limit=20').then(function(data) {
  state.sessions = data.items || data;
  renderHistory();
  if (state.sessions.length > 0 && !state.currentSessionId) {
    selectSession(state.sessions[0].id);
  }
}).catch(function(err) {
  // Fallback: use demo data if API unavailable
  state.sessions = demoSessions;
  renderHistory();
});
```

2. `doPlan()` → 替换 `setTimeout` 模拟为：
```js
// Auto-create session if needed
var createSession = state.currentSessionId
  ? Promise.resolve(state.currentSessionId)
  : EARP.fetchJSON('v1/sessions', {method:'POST'})
      .then(function(d){ state.currentSessionId = d.session_id; return d.session_id; });

createSession.then(function(sid) {
  return EARP.fetchJSON('plan', {
    method: 'POST',
    body: JSON.stringify({intent: intent, session_id: sid})
  });
}).then(function(data) {
  state.steps = data;
  setPlanLoading(false);
  renderSteps();
}).catch(function(err) {
  setPlanLoading(false);
  showError('Plan failed: ' + (err.message || 'Unknown error'));
});
```

3. `doInvoke(index)` → 替换为：
```js
var step = state.steps[index];
step.loading = true;
step.result = null;
step.error = null;
renderSteps();

EARP.fetchJSON('v1/sessions/' + state.currentSessionId + '/invoke', {
  method: 'POST',
  body: JSON.stringify({
    capability_call: {
      adapter_type: step.capability_id,
      input: step.input
    }
  })
}).then(function(data) {
  step.loading = false;
  step.result = data;
  renderSteps();
}).catch(function(err) {
  step.loading = false;
  step.error = err.message || 'Invoke failed';
  renderSteps();
});
```

**Phase 1 策略**：保留模拟数据作为 fallback，API 不可用时自动降级。确保页面在无后端环境下仍可演示交互。

## 风险

| 风险 | 缓解 |
|:---|:---|
| 后端 API 暂不可用 | Phase 1 用模拟数据 fallback，页面可独立演示 |
| `GET /v1/sessions` 不返回 `last_plan_*` | 前端用 fallback 缺省值（显示 "No plan yet"） |
| `POST /plan` 返回格式与预期不同 | 前端做防御性解析，catch 后显示 error banner |
